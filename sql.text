/* ---------- 10. FUNCTIONS  (call: dbo.fn_...) ---------- */

CREATE FUNCTION fn_GetAge (@dob DATE)
RETURNS INT
AS
BEGIN
    RETURN DATEDIFF(YEAR, @dob, GETDATE());
END;
GO

CREATE FUNCTION fn_GetGrade (@salary INT)
RETURNS VARCHAR(50)
AS
BEGIN
    DECLARE @grade VARCHAR(50);
    IF (@salary < 12000)
        SET @grade = 'C Grade';
    ELSE IF (@salary >= 12000 AND @salary <= 15000)
        SET @grade = 'B Grade';
    ELSE
        SET @grade = 'A Grade';
    RETURN @grade;
END;
GO

CREATE FUNCTION fn_AnnualSalary (@salary INT)
RETURNS INT
AS
BEGIN
    RETURN @salary * 12;
END;
GO

SELECT
    uid, name, gender, salary,
    dbo.fn_GetGrade(salary)     AS grade,
    dbo.fn_AnnualSalary(salary) AS annualsalary,
    dob,
    dbo.fn_GetAge(dob)          AS age
FROM tblusers;

CREATE PROCEDURE sp_GetAllUsers

GO

CREATE PROCEDURE sp_InsertUser
    @name    VARCHAR(50),
    @gender  VARCHAR(50),
    @salary  INT,
    @dob     DATE,
    @country INT
AS
BEGIN
    SET NOCOUNT ON;
    INSERT INTO tblusers (name, gender, salary, dob, country)
    VALUES (@name, @gender, @salary, @dob, @country);
    SELECT SCOPE_IDENTITY() AS NewUserID;
END;
GO

EXEC sp_InsertUser
    @name = 'Rohan', @gender = 'male', @salary = 22000,
    @dob = '1995-05-12', @country = 1;


CREATE PROCEDURE sp_GetUserStats
    @TotalUsers INT OUTPUT,
    @AvgSalary  INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT @TotalUsers = COUNT(*), @AvgSalary = AVG(salary) FROM tblusers;
END;
GO

DECLARE @UserCount INT, @AverageSal INT;
EXEC sp_GetUserStats @TotalUsers = @UserCount OUTPUT, @AvgSalary = @AverageSal OUTPUT;
SELECT @UserCount AS TotalUsers, @AverageSal AS AverageSalary;

ON tblusers

AS

CREATE TABLE tblSalaryHistory
(
    LogID        INT PRIMARY KEY IDENTITY,
    UserID       INT,
    OldSalary    INT,
    NewSalary    INT,
    ModifiedDate DATETIME
);
GO

CREATE TRIGGER trg_TrackSalaryChange
ON tblusers
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    IF UPDATE(salary)
    BEGIN
        INSERT INTO tblSalaryHistory (UserID, OldSalary, NewSalary, ModifiedDate)
        SELECT i.uid, d.salary, i.salary, GETDATE()
        FROM inserted i
        JOIN deleted  d ON i.uid = d.uid;
    END
END;
GO

SELECT * FROM tblSalaryHistory;

CREATE PROCEDURE moneytransfer
    @fromacc INT,
    @toacc   INT,
    @amt     INT
AS
BEGIN
    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE tblbank SET amount = amount - @amt WHERE accno = @fromacc;
        UPDATE tblbank SET amount = amount + @amt WHERE accno = @toacc;
        -- test fail:  WHERE accno = 'mohan'

        COMMIT TRANSACTION;
        PRINT 'Transaction Successful !!';
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT 'Transaction Failed !!';
    END CATCH
END;
GO
